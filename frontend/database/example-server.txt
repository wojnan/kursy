const express = require('express');
const { Pool } = require('pg');
const cors = require('cors');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');
require('dotenv').config();

const app = express();
app.use(cors());
app.use(express.json());

// Database connection
const pool = new Pool({
  connectionString: process.env.DATABASE_URL || 'postgresql://localhost:5432/learnhub'
});

// JWT middleware
function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];

  if (!token) return res.sendStatus(401);

  jwt.verify(token, process.env.JWT_SECRET || 'secret', (err, user) => {
    if (err) return res.sendStatus(403);
    req.user = user;
    next();
  });
}

  try {
    const result = await pool.query('SELECT * FROM courses ORDER BY id');

// ============================================
// SECTION & LESSON ENDPOINTS
// ============================================

app.get('/api/courses/:id/sections', async (req, res) => {
  try {
    const { id } = req.params;

    // Get sections
    const sectionsResult = await pool.query(
      'SELECT * FROM sections WHERE course_id = $1 ORDER BY order_index',
      [id]
    );

    // For each section, get lessons
    const sections = await Promise.all(sectionsResult.rows.map(async (section) => {
      const lessonsResult = await pool.query(
        'SELECT * FROM lessons WHERE section_id = $1 ORDER BY order_index',
        [section.id]
      );

      // For each lesson, get content
      const lessons = await Promise.all(lessonsResult.rows.map(async (lesson) => {
        const contentResult = await pool.query(
          'SELECT * FROM lesson_content WHERE lesson_id = $1 ORDER BY order_index',
          [lesson.id]
        );

        return {
          id: lesson.id.toString(),
          title: lesson.title,
          duration: lesson.duration,
          content: contentResult.rows.map(c => ({
            type: c.content_type,
            value: c.content_type === 'list' ? JSON.parse(c.content_value) : c.content_value
          }))
        };
      }));

      return {
        id: section.id.toString(),
        title: section.title,
        lessons
      };
    }));

    res.json(sections);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.get('/api/sections/:id/quiz', async (req, res) => {
  try {
    const { id } = req.params;

    const quizResult = await pool.query(
      'SELECT * FROM section_quizzes WHERE section_id = $1',
      [id]
    );

    if (quizResult.rows.length === 0) {
      return res.status(404).json({ error: 'Quiz not found' });
    }

    const quiz = quizResult.rows[0];

    const questionsResult = await pool.query(
      'SELECT * FROM quiz_questions WHERE section_quiz_id = $1 ORDER BY order_index',
      [quiz.id]
    );

    res.json({
      sectionId: quiz.section_id.toString(),
      sectionTitle: quiz.section_title,
      questions: questionsResult.rows.map(q => ({
        id: q.id.toString(),
        question: q.question,
        options: [q.option_1, q.option_2, q.option_3, q.option_4],
        correctAnswer: q.correct_answer
      }))
    });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.post('/api/auth/signup', async (req, res) => {
  try {
    const { email, password, name } = req.body;
    const hashedPassword = await bcrypt.hash(password, 10);

    const result = await pool.query(
      'INSERT INTO users (email, password_hash, name) VALUES ($1, $2, $3) RETURNING id, email, name, created_at',
      [email, hashedPassword, name]
    );

    const user = result.rows[0];
    const token = jwt.sign({ id: user.id, email: user.email }, process.env.JWT_SECRET || 'secret');

    res.json({ user, token });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.post('/api/auth/login', async (req, res) => {
  try {
    const { email, password } = req.body;

    const result = await pool.query('SELECT * FROM users WHERE email = $1', [email]);
    if (result.rows.length === 0) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }

    const user = result.rows[0];
    const validPassword = await bcrypt.compare(password, user.password_hash);

    if (!validPassword) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }

    const token = jwt.sign({ id: user.id, email: user.email }, process.env.JWT_SECRET || 'secret');

    res.json({
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        created_at: user.created_at
      },
      token
    });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// PROGRESS ENDPOINTS
// ============================================

app.get('/api/users/:id/progress', authenticateToken, async (req, res) => {
  try {
    const { id } = req.params;
    const { course_id } = req.query;

    let query = 'SELECT * FROM user_progress WHERE user_id = $1';
    const params = [id];

    if (course_id) {
      query += ' AND section_id IN (SELECT id FROM sections WHERE course_id = $2)';
      params.push(course_id);
    }

    const result = await pool.query(query, params);
    res.json(result.rows);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.post('/api/progress/section', authenticateToken, async (req, res) => {
  try {
    const { user_id, section_id } = req.body;

    const result = await pool.query(
      'INSERT INTO user_progress (user_id, section_id) VALUES ($1, $2) ON CONFLICT DO NOTHING RETURNING *',
      [user_id, section_id]
    );

    res.json(result.rows[0] || { user_id, section_id });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});
